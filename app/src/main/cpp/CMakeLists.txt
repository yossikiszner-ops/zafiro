cmake_minimum_required(VERSION 3.22.1)
project(zafiro_android_brain LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llama
    GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git
    GIT_TAG 03aa006acb547549162a00dddeab3d3b38ebf7b2)
FetchContent_MakeAvailable(llama)
add_library(zafiro_android_brain SHARED android_brain.cpp)
target_compile_features(zafiro_android_brain PRIVATE cxx_std_17)
target_link_libraries(zafiro_android_brain PRIVATE llama mtmd)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(zafiro_android_brain PRIVATE ${JNI_INCLUDE_DIRS})
endif()
